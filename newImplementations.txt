// App.tsx
import React, { useState } from 'react';
import ReactFlow, { Background, Controls } from 'react-flow-renderer';
import { parsePromptToFlowchart } from './utils/parsePromptToNodes';

export default function App() {
  const [elements, setElements] = useState({ nodes: [], edges: [] });
  const [prompt, setPrompt] = useState('');

  const generate = () => {
    const flow = parsePromptToFlowchart(prompt);
    setElements(flow);
  };

  return (
    <div style={{ width: '100vw', height: '100vh' }}>
      <div style={{ padding: 20 }}>
        <input
          style={{ width: 400 }}
          placeholder="Enter your idea..."
          value={prompt}
          onChange={(e) => setPrompt(e.target.value)}
        />
        <button onClick={generate}>Generate Diagram</button>
      </div>
      <ReactFlow nodes={elements.nodes} edges={elements.edges} fitView>
        <Background />
        <Controls />
      </ReactFlow>
    </div>
  );
}

System Prompt

Copy
Edit
You are a diagram assistant. Your job is to turn user questions or ideas into a structured flowchart. 

Return a JSON object with two keys:
- "nodes": list of labeled steps or decisions
- "edges": list of connections between node IDs, optionally with labels (like "Yes"/"No")

Each node should have:
  - id (string)
  - label (string)
  - type (optional: "default", "input", "output", or "decision")

Each edge should have:
  - id (string)
  - source (string, node id)
  - target (string, node id)
  - label (optional)

Do not explain anything. Just return valid JSON.

// utils/layoutGraph.ts
import dagre from 'dagre';
import { Node, Edge } from 'react-flow-renderer';

const nodeWidth = 180;
const nodeHeight = 60;

export function layoutNodesWithDagre(
  nodes: Node[],
  edges: Edge[],
  direction: 'TB' | 'LR' = 'TB' // Top-Bottom or Left-Right
): { nodes: Node[]; edges: Edge[] } {
  const dagreGraph = new dagre.graphlib.Graph();
  dagreGraph.setDefaultEdgeLabel(() => ({}));
  dagreGraph.setGraph({ rankdir: direction });

  nodes.forEach((node) => {
    dagreGraph.setNode(node.id, { width: nodeWidth, height: nodeHeight });
  });

  edges.forEach((edge) => {
    dagreGraph.setEdge(edge.source, edge.target);
  });

  dagre.layout(dagreGraph);

  const positionedNodes = nodes.map((node) => {
    const { x, y } = dagreGraph.node(node.id);
    return {
      ...node,
      position: { x, y },
      sourcePosition: direction === 'LR' ? 'right' : 'bottom',
      targetPosition: direction === 'LR' ? 'left' : 'top',
    };
  });

  return { nodes: positionedNodes, edges };
}

Copy
Edit
import { layoutNodesWithDagre } from './utils/layoutGraph';

setElements({ nodes, edges });
